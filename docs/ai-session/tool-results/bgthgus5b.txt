    41	def severe_range(hz: dict) -> tuple[float, float]:
    42	    """Conservative 90% range for severe days: (warning-days low x share low, warning-days high x share high).
    43	    Combining the two intervals this way is wider than a joint interval, i.e. it errs on the cautious side."""
    44	    w_lo, w_hi = (x or 0.0 for x in hz["warning_days_range"])
    45	    s = hz["severity_share"]
    46	    return w_lo * s["lo"], w_hi * s["hi"]

    49	def _latest(conn: sqlite3.Connection) -> pd.DataFrame:
    50	    df = pd.read_sql(
    51	        "SELECT r.hub_id, h.name, h.state, h.region, r.composite, r.winter, r.flood, r.convective, r.hurricane, "
    52	        "r.heat, r.tier, r.scoring_version, r.computed_at, r.details_json FROM risk_scores r JOIN hubs h "
    53	        "ON h.id = r.hub_id WHERE r.computed_at = (SELECT MAX(computed_at) FROM risk_scores)", conn)
    54	    details = df["details_json"].map(json.loads)
    55	    for h in HAZARDS:
    56	        df[f"routine_{h}"] = details.map(lambda d, h=h: d["hazards"][h]["warning_days_per_year"] or 0.0)
    57	        df[f"routine_{h}_lo"] = details.map(lambda d, h=h: d["hazards"][h]["warning_days_range"][0] or 0.0)
    58	        df[f"routine_{h}_hi"] = details.map(lambda d, h=h: d["hazards"][h]["warning_days_range"][1] or 0.0)
    59	        lo, hi = zip(*details.map(lambda d, h=h: severe_range(d["hazards"][h])))
    60	        df[f"{h}_lo"], df[f"{h}_hi"] = list(lo), list(hi)
    61	    df["routine_total"] = details.map(lambda d: d["routine_total"])
    62	    df["confidence"] = details.map(lambda d: d["confidence"])
    63	    return df.drop(columns="details_json")

    66	def _with_conn(fn):
    67	    def wrapper(args, conn=None):
    68	        own = conn is None
    69	        conn = conn or get_connection()
    70	        try:
    71	            return fn(args, conn)
    72	        finally:
    73	            if own:
    74	                conn.close()
    75	    wrapper.__name__, wrapper.__doc__ = fn.__name__, fn.__doc__

   273	@_with_conn
   274	def get_weather_stat(args: s.WeatherStatInput, conn) -> dict:
   275	    """Day counts / totals for any date range.
   276	
   277	    Gap filling differs on purpose from the scoring engine: for an arbitrary date range,
   278	    filling day by day keeps as much measured data as possible; the engine's whole-variable
   279	    fallback is about consistency over a 30-year window. The output states which was used.
   280	    """
   281	    hub, metric = args.hub_id.value, args.metric
   282	    start, end = args.start_date.isoformat(), args.end_date.isoformat()
   283	    days = pd.date_range(start, end).strftime("%Y-%m-%d")
   284	    base = {"hub_id": hub, "metric": metric, "start_date": start, "end_date": end, "days_in_period": len(days)}
   285	
   286	    if metric.startswith("warning_days_"):
   287	        hazard = metric.removeprefix("warning_days_")
   288	        wp = get_params()["warnings"]
   289	        ugc = conn.execute("SELECT ugc, timezone FROM hub_ugc WHERE hub_id=?", (hub,)).fetchall()
   290	        codes = [u[0] for u in ugc] + [f"POINT:{hub}"]
   291	        w = pd.read_sql(f"SELECT phenomena, issue, expire FROM nws_warnings WHERE ugc IN ({','.join('?' * len(codes))})",
   292	                        conn, params=codes)
   293	        wd = m.warning_days(w, ugc[0][1], wp["phenomena"], (wp["first_year"], wp["last_year"]))[hazard]
   294	        lo_ok, hi_ok = f"{wp['first_year']}-01-01", f"{wp['last_year']}-12-31"
   295	        covered = [d for d in days if lo_ok <= d <= hi_ok]
   296	        count = sum(d in wd for d in covered)
   297	        return {**base, "count": count, "days_with_data": len(covered),
   298	                "percent_of_days": _r(100 * count / len(covered)) if covered else None,
   299	                "definition": f"days with an NWS warning of types {wp['phenomena'][hazard]} at the hub",
   300	                "source": f"National Weather Service warnings ({wp['first_year']}-{wp['last_year']})",
   301	                "caveat": None if len(covered) == len(days) else "part of the period is outside the warning archive window"}
   302	
   303	    st = conn.execute("SELECT station_id, station_name, distance_km, gust_measured FROM hub_stations WHERE hub_id=?",
   304	                      (hub,)).fetchone()
   305	    station = pd.read_sql("SELECT * FROM station_daily WHERE station_id=? AND date BETWEEN ? AND ?", conn,
   306	                          params=(st[0], start, end)).set_index("date") if st else pd.DataFrame()
   307	    om = pd.read_sql("SELECT * FROM daily_weather WHERE hub_id=? AND date BETWEEN ? AND ?", conn,
   308	                     params=(hub, start, end)).set_index("date")
   309	    st_info = {"station": st[1], "station_id": st[0], "distance_km": st[2]} if st else None
   310	
   311	    if metric in ("ice_day", "thunder_day"):
   312	        if not st or not st[3]:
   313	            return {**base, "count": None, "error": "This hub's station does not record weather events; "
   314	                                                    "ice/thunder days are unavailable."}
   315	        cols = ["wt_glaze", "wt_freezing_rain", "wt_ice_pellets"] if metric == "ice_day" else ["wt_thunder"]
   316	        obs = station.reindex(days)
   317	        has = obs["prcp_mm"].notna() | obs["tmax_c"].notna()       # station reported that day
   318	        count = int((obs.loc[has, cols].fillna(0).sum(axis=1) > 0).sum())
   319	        n = int(has.sum())
   320	        return {**base, "count": count, "days_with_data": n, "percent_of_days": _r(100 * count / n) if n else None,
   321	                "definition": "observed " + ("glaze, freezing rain or ice pellets" if metric == "ice_day" else "thunder"),
   322	                "source": "Weather events observed at the airport station (NOAA)",
   323	                **({"station": st_info} if st_info else {}),
   324	                "caveat": "Flags are recorded inconsistently; a missing flag doesn't prove the event didn't happen."}
   325	
   326	    var = DAY_FLAGS[metric][0] if metric in DAY_FLAGS else TOTALS[metric]
   327	    st_col, om_col = m.VARIABLES[var]
   328	    st_s = station[st_col].reindex(days) if st_col in station else pd.Series(index=days, dtype=float)
   329	    if var == "gust_kmh" and st and not st[3]:
   330	        st_s = pd.Series(index=days, dtype=float)               # no measured gusts at this station
   331	    om_s = om[om_col].reindex(days) if om_col in om else pd.Series(index=days, dtype=float)
   332	    combined = st_s.combine_first(om_s)
   333	    n = int(combined.notna().sum())
   334	    sources = {"station_days": int(st_s.notna().sum()), "open_meteo_days": int((st_s.isna() & om_s.notna()).sum())}
   335	    res = {**base, "days_with_data": n, "sources": sources, "method": GAP_FILL_METHOD,
   336	           **({"station": st_info} if st_info else {})}
   337	    if metric in TOTALS:
   338	        res.update({"total": _r(combined.sum(), 1), "unit": "mm" if var == "precip_mm" else "cm"})
   339	    else:
   340	        count = int(combined.dropna().map(DAY_FLAGS[metric][1]).sum())
   341	        res.update({"count": count, "percent_of_days": _r(100 * count / n) if n else None,
   342	                    "definition": DAY_DEFINITIONS[metric]})
   343	    res["source"] = ("Measured weather at the airport station (NOAA)" if sources["open_meteo_days"] == 0 else
   344	                     "Measured weather at the airport station (NOAA), with a few missing days estimated from "

   403	@_with_conn
   404	def get_seasonal_outlook(args: s.SeasonalOutlookInput, conn) -> dict:
   405	    """Which hubs historically face the most disruption in a given calendar month, and why."""
   406	    month = args.month or date.today().month
   407	    q = ("SELECT m.*, h.name, h.state, h.region, h.county_fips FROM hub_monthly m JOIN hubs h ON h.id = m.hub_id "
   408	         "WHERE m.month = ?" + (" AND h.region = ?" if args.region else ""))
   409	    df = pd.read_sql(q, conn, params=(month, args.region) if args.region else (month,))
   410	    totals = df.groupby("hub_id")["severe_days"].sum().round(6)
   411	    rank = totals.rank(ascending=False, method="min").astype(int)
   412	    order = sorted(totals.index, key=lambda h: (rank[h], h))[: (args.top_n or 5)]
   413	    decl = pd.read_sql("SELECT county_fips, title, incident_type, begin_date FROM fema_declarations", conn)
   414	    decl = decl[decl["begin_date"].str[5:7] == f"{month:02d}"].sort_values("begin_date", ascending=False)
   415	
   416	    hubs = []
   417	    for hub in order:
   418	        rows = df[df["hub_id"] == hub].set_index("hazard")
   419	        top = rows["severe_days"].idxmax() if rows["severe_days"].max() > 0 else rows["mean_warning_days"].idxmax()
   420	        t = rows.loc[top]
   421	        other = [{"hazard": z, "p_at_least_one_warning_day": _r(rows.loc[z, "p_warning"], 2)}
   422	                 for z in rows.index if z != top and rows.loc[z, "p_warning"] >= 0.3]
   423	        past = decl[decl["county_fips"] == t["county_fips"]].head(3)
   424	        item = {
   425	            "rank": int(rank[hub]), "hub_id": hub, "name": t["name"], "state": t["state"], "region": t["region"],
   426	            "expected_severe_days_this_month": _r(totals[hub], 3),
   427	            "top_hazard": top, "top_hazard_label": HAZARD_LABELS[top],
   428	            "p_at_least_one_warning_day": _r(t["p_warning"], 2),
   429	            "p_range_90pct": [_r(t["p_lo"], 2), _r(t["p_hi"], 2)],
   430	            "years_with_warning": int(t["years_with_warning"]), "years_observed": int(t["years"]),
   431	            "avg_warning_days_this_month": _r(t["mean_warning_days"], 2),
   432	            "other_likely_hazards": other,
   433	            "past_fema_disasters_this_month": [{"title": r.title, "type": r.incident_type, "year": r.begin_date[:4]}
   434	                                               for r in past.itertuples()],
   435	        }
   436	        if top == "hurricane" and pd.notna(t["longrecord_p"]):
   437	            item["p_storm_within_150km_1950_2025"] = _r(t["longrecord_p"], 2)
   438	        hubs.append(item)
   439	
   440	    summary = {}
   441	    for z in HAZARDS:
   442	        zr = df[df["hazard"] == z]
   443	        likely = zr[zr["p_warning"] >= 0.5]
   444	        best = zr.sort_values(["p_warning", "hub_id"], ascending=[False, True]).head(1)
   445	        summary[z] = {"hubs_with_warning_in_most_years": int(len(likely)),
   446	                      "most_frequent_hub": None if best.empty or best["p_warning"].iloc[0] == 0 else best["hub_id"].iloc[0],
   447	                      "its_p": None if best.empty else _r(best["p_warning"].iloc[0], 2)}
   448	    return {"month": month, "month_name": MONTHS[month - 1], "region": args.region, "hubs": hubs,
   449	            "hazard_summary": summary,
   450	            "definition": "p_at_least_one_warning_day = share of years 2006-2025 in which the hub had at least one "
   451	                          "NWS warning day for the hazard in this month; expected_severe_days_this_month = average "
   452	                          "warning days in this month x the hazard's FEMA severity share, summed over hazards.",
   453	            "note": "Historical frequency (climatology), not a forecast.",
   454	            "source": SOURCES}
